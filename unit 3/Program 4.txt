SET SERVEROUTPUT ON;

DECLARE
    V_SAL EMP.SAL%TYPE;
BEGIN
    SELECT SAL
    INTO V_SAL
    FROM EMP
    WHERE AGE = 50;

    DBMS_OUTPUT.PUT_LINE('Salary of employee whose age is 50: ' || V_SAL);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No employee found whose age is 50.');
    WHEN TOO_MANY_ROWS THEN
        DBMS_OUTPUT.PUT_LINE('More than one employee is 50 years old.');
END;
/
