SET SERVEROUTPUT ON;

DECLARE
    V_NUM1 NUMBER := 50;
    V_NUM2 NUMBER := 0;
    V_RESULT NUMBER;
BEGIN
    V_RESULT := V_NUM1 / V_NUM2;

    DBMS_OUTPUT.PUT_LINE('Result = ' || V_RESULT);

EXCEPTION
    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE(
            'ZERO_DIVIDE: Division by zero is not allowed.'
        );
END;
/
