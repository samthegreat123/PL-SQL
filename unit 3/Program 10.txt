SET SERVEROUTPUT ON;

DECLARE
    V_NAME EMP.ENAME%TYPE;
BEGIN
    SELECT ENAME
    INTO V_NAME
    FROM EMP
    WHERE EMPNO = 9999;

    DBMS_OUTPUT.PUT_LINE('Employee Name: ' || V_NAME);

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error Code    : ' || SQLCODE);
        DBMS_OUTPUT.PUT_LINE('Error Message : ' || SQLERRM);
END;
/
